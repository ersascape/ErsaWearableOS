9// Semantic companion events are transport-neutral. Providers may be backed by
10// ANCS/AMS over BLE, an Android companion protocol, Linux MPRIS, or a test.
11// Callback string pointers are borrowed for the callback duration only; sinks
12// must copy them before returning. Sources may call callbacks from any thread.
13enum class CompanionCallAction : uint8_t { Incoming = 0, Answered, Rejected, Ended };
14enum class CompanionMediaAction : uint8_t { Play = 0, Pause, Toggle, Next, Previous, VolumeUp, VolumeDown };

16using CompanionCallCallback = void (*)(CompanionCallAction, const char*, const char*, void*);
17using CompanionMediaCallback = void (*)(bool, const char*, const char*, void*);
18using CompanionTimeCallback = void (*)(uint32_t, void*);
19using CompanionAvailabilityCallback = void (*)(bool, void*);
20// Null title/message remove one UID; null app as well clears the provider session.
